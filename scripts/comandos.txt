# ========================================================
# 🤖 GUÍA DE COMANDOS - VENDEDOR IA (n8n + PostgreSQL)
# ========================================================

# --------------------------------------------------------
# 1. GESTIÓN DE DOCKER (Contenedores)
# --------------------------------------------------------
# Encender el sistema en segundo plano (Recomendado)
docker compose up -d

# Ver en tiempo real qué está pasando (Logs)
docker compose logs -f

# Apagar el sistema
docker compose down

# Apagar y borrar toda la base de datos (¡CUIDADO!)
docker compose down -v


# --------------------------------------------------------
# 2. IMPORTAR FLUJOS A N8N DESDE LA TERMINAL
# --------------------------------------------------------
# Importar el flujo principal (flujo.json)
docker compose exec n8n n8n import:workflow --input=/data/flujo.json

# Importar el flujo de seguimiento (flujo_seguimiento.json)
docker compose exec n8n n8n import:workflow --input=/data/flujo_seguimiento.json


# --------------------------------------------------------
# 3. BASE DE DATOS (POSTGRESQL)
# --------------------------------------------------------
# Entrar a la consola SQL interactiva de Postgres
# (Una vez adentro, puedes escribir comandos como: SELECT * FROM clientes;)
docker compose exec db psql -U n8n -d vendedor_ia

# Hacer un backup manual de la base de datos a un archivo
docker compose exec db pg_dump -U n8n -d vendedor_ia > backup.sql

# Restaurar un backup
cat backup.sql | docker compose exec -T db psql -U n8n -d vendedor_ia


# --------------------------------------------------------
# 4. SOLUCIÓN DE PROBLEMAS COMUNES
# --------------------------------------------------------
# Error: "Mismatching encryption keys" (Ocurre si pierdes tu .env)
# Solución: Borrar el archivo de config de n8n para que genere uno nuevo.
# Nota: Tendrás que volver a poner tus contraseñas en los nodos de n8n.
rm n8n_data/config
docker compose restart n8n
